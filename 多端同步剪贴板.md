//! mDNS service discovery for PasteBridge multi-device sync.
//!
//! - Registers this device under `_pastebridge._tcp.local.` on the given port
//! - Browses for other devices running PasteBridge on the LAN

pub mod discovery {
    use std::collections::HashMap;
    use std::sync::{Arc, Mutex};
    use std::time::Duration;
    use mdns_sd::{ServiceDaemon, ServiceEvent, ServiceInfo};

    pub const SERVICE_TYPE: &str = "_pastebridge._tcp.local.";

    #[derive(Debug, Clone)]
    pub struct DiscoveredPeer {
        pub device_id: String,
        pub platform: String,
        pub addresses: Vec<String>,
        pub port: u16,
    }

    pub struct Discovery {
        daemon: ServiceDaemon,
        registered_fullname: Mutex<Option<String>>,
    }

    impl Discovery {
        pub fn new() -> Result<Self, String> {
            let daemon = ServiceDaemon::new().map_err(|e| format!("Failed to create mDNS daemon: {}", e))?;
            Ok(Self {
                daemon,
                registered_fullname: Mutex::new(None),
            })
        }

        pub fn register(&self, device_id: &str, platform: &str, port: u16) -> Result<(), String> {
            let host = get_hostname();
            let service_name = format!("PasteBridge-{}", &device_id[..device_id.len().min(8)]);
            
            let mut properties = HashMap::new();
            properties.insert("device_id".to_string(), device_id.to_string());
            properties.insert("platform".to_string(), platform.to_string());

            let service_info = ServiceInfo::new(
                SERVICE_TYPE,
                &service_name,
                &host,
                port,
                properties,
            ).map_err(|e| format!("Failed to build ServiceInfo: {}", e))?;

            let fullname = service_info.get_fullname().to_string();
            self.daemon.register(service_info)
                .map_err(|e| format!("Failed to register mDNS service: {}", e))?;
            *self.registered_fullname.lock().unwrap() = Some(fullname.clone());
            eprintln!("[mdns] Registered: {} @ :{}", fullname, port);
            Ok(())
        }

        pub fn browse<F>(&self, on_discovered: F) -> Result<(), String>
        where
            F: Fn(DiscoveredPeer) + Send + 'static,
        {
            let receiver = self.daemon.browse(SERVICE_TYPE)
                .map_err(|e| format!("Failed to start browse: {}", e))?;
            
            std::thread::spawn(move || {
                while let Ok(event) = receiver.recv() {
                    match event {
                        ServiceEvent::ServiceResolved(info) => {
                            let name = info.get_fullname().to_string();
                            let port = info.get_port();
                            let mut device_id = String::new();
                            let mut platform = String::new();
                            
                            for prop in info.get_properties().iter() {
                                let key = prop.key();
                                let val = prop.val_str();
                                if key == "device_id" {
                                    device_id = val.to_string();
                                } else if key == "platform" {
                                    platform = val.to_string();
                                }
                            }
                            
                            if device_id.is_empty() {
                                // fallback: extract from name
                                device_id = info.get_name().to_string();
                            }
                            
                            let addresses: Vec<String> = info.get_addresses()
                                .iter()
                                .map(|a| a.to_string())
                                .collect();
                            
                            let peer = DiscoveredPeer {
                                device_id: device_id.clone(),
                                platform,
                                addresses: addresses.clone(),
                                port,
                            };
                            
                            eprintln!("[mdns] Discovered: {} @ {:?}:{} (platform={})",
                                device_id, addresses, port, peer.platform);
                            on_discovered(peer);
                        }
                        ServiceEvent::SearchStopped(_) => {
                            eprintln!("[mdns] Browse stopped");
                            break;
                        }
                        _ => {}
                    }
                }
            });
            
            Ok(())
        }

        pub fn shutdown(&self) -> Result<(), String> {
            if let Some(fullname) = self.registered_fullname.lock().unwrap().take() {
                let _ = self.daemon.unregister(&fullname);
            }
            let _ = self.daemon.shutdown();
            Ok(())
        }
    }

    fn get_hostname() -> String {
        std::env::var("COMPUTERNAME")
            .or_else(|_| std::env::var("HOSTNAME"))
            .unwrap_or_else(|_| "pastebridge-host".to_string())
    }
}

### 环境约束

mDNS 用 224.0.0.251 链路层组播，**必须**两端在同一 L2 域（Wi-Fi/以太网/桥接）。Internet / 蜂窝不通。

### 这一轮做的事情

1. **APK 装机**：`adb install` Success,设备 Android 16 / API 36（和 compileSdk 完全匹配）
2. **第一次启动**：进程立刻崩,FATAL EXCEPTION:
   ```
   java.lang.SecurityException: NsdService: Neither user 10983 nor current process has android.permission.INTERNET.
   ```
   **Root cause**:`AndroidManifest.xml` 缺 `android.permission.INTERNET`。Android 14+ 强制要求 NsdManager 显式声明。
3. **修复**：补 4 个权限到 `AndroidManifest.xml`:
   - `INTERNET` (必需,NSD 内部 socket 通信)
   - `ACCESS_NETWORK_STATE` (NSD 检查网络可达性)
   - `ACCESS_WIFI_STATE` / `CHANGE_WIFI_MULTICAST_STATE` (部分 ROM 接收组播需要)
4. **重编重装**：再次启动 ✅ 不崩
5. **adb logcat 验证**:
   - `D PBDiscovery: onDiscoveryStarted: _pastebridge._tcp.`  ← browse 启动成功
   - `I serviceDiscovery: [MdnsAdvertiser] Adding service name: PasteBridge-android-smok-18792, type: _pastebridge._tcp, ...txtRecord: {device_id=android-smoke, platform=android}`  ← 系统层 NSD 已接受注册
   - 注册回调 `onServiceRegistered` 没触发:因为 `hostAddresses: , hostname: null, network: null` (没组播网络)
   - 浏览回调 `onServiceResolved` 没触发:同原因
6. **进程稳定性**：PID 16471 持续存活,无崩溃

### 仍遗留的小问题

`W PBDiscovery: browse called while already browsing, skipping` — `DiscoveryBanner` Composable 每次重组会再调一次 `browse`。`DiscoveryService.browse()` 内部已做了 idempotent 保护 (发现 `discoveryListener != null` 就 return),不阻塞功能。**仅是 warning,后续清理。**
